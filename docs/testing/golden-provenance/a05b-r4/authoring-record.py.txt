def read(path):
    return json.loads(path.read_text(encoding="utf-8"))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def canonical(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":"),
                                     ensure_ascii=True).encode()).hexdigest()


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")

quantum = Decimal(".0001")

# Re-reduce retained independent terms, without importing a production engine.
worksheet = read(OLD / "fundamental-derivation.json")
parts = {}
for name, component in worksheet["components"].items():
    numerator = sum(Decimal(t["normalized"]) * Decimal(t["weight"]) for t in component["terms"])
    denominator = sum(Decimal(t["weight"]) for t in component["terms"])
    reduced = max(Decimal(0), min(Decimal(10), numerator / denominator)).quantize(quantum)
    reduced = max(Decimal(0), min(Decimal(10), reduced + Decimal(component["bonus"]))).quantize(quantum)
    assert reduced == Decimal(component["expected"])
    parts[name] = str(reduced)
fundamental = sum(Decimal(parts[k]) * Decimal(str(w)) for k, w in worksheet["weights"].items())
fundamental = fundamental.quantize(quantum, rounding=ROUND_HALF_EVEN)
assert fundamental == Decimal(worksheet["expected"])

technical = {r["ticker"]: r for r in core["technical"]}
combined = {r["ticker"]: r for r in core["combined"]}
reductions = []
for ticker, terms in ledger["technical"].items():
    source = technical[ticker]
    assert source["warning_flags_json"] == [] and source["technical_confidence"] == "high"
    # READY -> ELIGIBLE is shared by v1/v2. The research-only ceiling predicate is false.
    composite = sum(float(terms["v4_parts"][key]) * float(weight)
                    for key, weight in terms["regime_weights"].items())
    dual = round(composite, 4)
    assert Decimal(str(dual)) == Decimal(source["dual_score"]), (ticker, dual)
    penalty = Decimal("1.5") if source["classification"] == "Overheated momentum" else Decimal(0)
    score = Decimal(str(round(float(fundamental) * .55 + float(dual) * .45 - float(penalty), 4)))
    assert score == Decimal(combined[ticker]["final_score"]), ticker
    label = "Strong candidate" if score >= 8 else "Candidate" if score >= Decimal("6.8") else "Watchlist"
    assert label == combined[ticker]["combined_decision"]
    reductions.append({"ticker": ticker, "technical_composite": str(dual),
                       "combined_score": str(score), "combined_label": label,
                       "research_only": False, "warnings": []})
ranking = {(r["profile"], r["ticker"]): r for r in core["ranking"]}
ranking_reductions = []
for terms in ledger["ranking_arithmetic"]:
    profile, ticker = terms["profile"], terms["ticker"]
    independently_summed = sum(float.fromhex(x) for x in terms["binary64_terms_hex"])
    assert independently_summed.hex() == terms["binary64_sum_hex"]
    assert round(independently_summed, 4) == float(terms["technical"])
    row = ranking[profile, ticker]
    penalty = sum(float(x) for x in row["penalties"].values())
    final = round(float.fromhex(terms["weighted_binary64_hex"]) - penalty, 4)
    assert Decimal(str(final)) == Decimal(row["profile_score"])
    ranking_reductions.append({"profile": profile, "ticker": ticker,
                               "technical": str(round(independently_summed, 4)),
                               "penalty": str(penalty), "score": str(final)})
