"""Independent CERI worksheet for the frozen replay, including valid absences."""
from datetime import date
from derive_core import DEST, D, dump, load, tickers

replay = load("provider-replay.json")
estimates, guidance, catalysts, revisions, derived, prices, captures = [], [], [], [], [], [], []
derivations = {}
for ticker in tickers:
    rows = [row for row in replay["eodhd"]["estimates"] if row["ticker"] == ticker]
    rows.sort(key=lambda row: row["observed_at"])
    current = rows[-1]
    assert len(rows) == 4 and all(row["currency"] == "USD" for row in rows)
    for row in rows:
        estimates.append({"ticker": ticker, "record_key": row["provider_record_id"],
                          "metric": "EPS_DILUTED", "period_type": "CURRENT_FISCAL_YEAR",
                          "fiscal_period_end": "2026-12-31", "canonical_currency": "USD",
                          "currency_verified": True, "canonical_scale": "1",
                          **{key: row[key] for key in ("consensus", "high", "low", "analyst_count", "upward_count", "downward_count")},
                          "known_at": row["retrieved_at"], "effective_session": row["observed_at"][:10]})
    values = []
    for window in (7, 30, 90):
        # Policy chooses newest prior observation at or before cutoff-window,
        # excluding current. Tolerance only applies when none precede target.
        target = date(2026, 10, 26).toordinal()-window
        baseline = max((row for row in rows[:-1] if date.fromisoformat(row["observed_at"][:10]).toordinal() <= target),
                       key=lambda row: row["observed_at"])
        elapsed = (date.fromisoformat(current["observed_at"][:10])-date.fromisoformat(baseline["observed_at"][:10])).days
        absolute = D(current["consensus"])-D(baseline["consensus"])
        pct = absolute/abs(D(baseline["consensus"]))*100
        row = {"ticker": ticker, "metric": "EPS_DILUTED", "period_slot": "CURRENT_FISCAL_YEAR",
               "window_days": window, "as_of_session": "2026-10-26", "actual_elapsed_days": elapsed,
               "absolute_change": absolute, "pct_change": pct, "acceleration": None,
               "net_breadth": D(7)/9, "dispersion": (D(current["high"])-D(current["low"]))/D(current["consensus"]),
               "revision_confidence_score": "8", "revision_confidence_label": "High",
               "unavailable_reason": None, "comparison_mode": "ABSOLUTE_CANONICAL",
               "current_record_key": current["provider_record_id"], "baseline_record_key": baseline["provider_record_id"],
               "warnings": [], "ownership_mode": "PIPELINE", "cutoff_at": "2026-10-26T20:32:00+00:00"}
        values.append(row)
    values[0]["acceleration"] = values[0]["pct_change"]/values[0]["actual_elapsed_days"]-(
        values[-1]["pct_change"]/values[-1]["actual_elapsed_days"])
    # Native revision columns persist six decimal places before the capture
    # job reads them in a fresh session. Keep that explicit storage boundary.
    for value in values:
        for key in ("absolute_change", "pct_change", "acceleration", "net_breadth", "dispersion"):
            if value[key] is not None:
                value[key] = value[key].quantize(D(".000001"))
    revisions.extend(values)
    guidance.append({"ticker": ticker, "record_key": ticker+"-sec-guidance", "action": "UNKNOWN",
                     "confidence": "INSUFFICIENT", "accepted_for_scoring": False,
                     "rejection_reason": "GUIDANCE_ACTION_UNKNOWN", "metric": "REVENUE",
                     "period_type": "CURRENT_FISCAL_YEAR", "effective_session": "2026-09-23",
                     "point_value": None,  # provider_storage_projection permits point_value, not replay key point
                     "supersedes": None})
    catalysts.append({"ticker": ticker, "record_key": ticker+"-catalyst", "category": "PRODUCT",
                      "subject_key": "launch", "revision_number": 1, "issuer_relevance": None,
                      "status": "ANNOUNCED", "direction": "POSITIVE", "materiality": ".7",
                      "date_confidence": "EXACT_TIMESTAMP", "source_confidence": "Normal",
                      "effective_session": "2026-09-23", "expected_date": "2026-10-15",
                      "prior_revision": None, "selected": False, "binary_eligible": None, "feature_binary_eligible": False,
                      "opportunity_available": False, "opportunity_component": "0", "binary_risk_score": "0",
                      "rejection_reason": "ISSUER_RELEVANCE_UNVERIFIED"})
    derived.extend([{"ticker": ticker, "feature_family": family, "feature_key": key,
                     "as_of_session": "2026-10-26", "ownership_mode": "PIPELINE"}
                    for family, key in (("catalysts", "current"), ("confidence", "score"))])
    prices.append({"ticker": ticker, "event_type": "NONE", "event_id": None, "quality": None,
                   "unavailable_reason": "NO_ACCEPTED_EVENT", "reaction_session": None,
                   "price_members": [], "ownership_mode": "PIPELINE"})
    components = {"revision_magnitude": sum(row["pct_change"] for row in values)/3,
                  "revision_breadth": (values[0]["net_breadth"]+1)*5,
                  "revision_acceleration": min(D(10), D(5)+values[0]["acceleration"]*10),
                  "surprise_trend": None, "guidance": None, "catalysts": None, "price_response": None}
    confidence_parts = {"source_quality": D(8), "freshness": None, "estimate_coverage": D("1.25"),
                        "analyst_sample": D(10), "timestamp_quality": D(9), "conflict_free_score": D(10)}
    confidence = D(8)*D(".25")+D("1.25")*D(".2")+D(10)*D(".15")+D(9)*D(".1")+D(10)*D(".1")
    assert confidence == D("5.65")
    captures.append({"ticker": ticker, "as_of_session": "2026-10-26", "cutoff_at": "2026-10-26T20:32:00+00:00",
                     "opportunity_score": None, "opportunity_coverage_pct": "50",
                     "opportunity_unrated_reason": "INSUFFICIENT_COMPONENT_COVERAGE",
                     "event_risk_score": "0", "earnings_proximity_risk": "0",
                     "data_confidence": "Low", "pipeline_ownership_state": "UNKNOWN", "confidence_score": confidence, "coverage_pct": "12.5",
                     "posture": "Unrated", "components": components,
                     "comparison_state": "NO_PRIOR_COMPARABLE_SNAPSHOT",
                     "warnings": ["surprise_trend_unavailable", "guidance_rows_rejected", "guidance_unavailable", "catalysts_unavailable",
                                  "price_response_unavailable", "opportunity_component_coverage_insufficient",
                                  "earnings_date_unavailable", "estimate_coverage_low", "dataset_freshness_unavailable"]})
    derivations[ticker] = {"baseline_keys": {row["window_days"]: row["baseline_record_key"] for row in values},
                           "revision_formula": "100*(current-baseline)/abs(baseline)",
                           "acceleration_formula": "pct7/elapsed7-pct90/elapsed90; other windows null",
                           "confidence_ledger": confidence_parts, "confidence_formula": "8*.25+0+1.25*.20+10*.15+9*.10+10*.10=5.65",
                           "ownership_policy": "post-acquisition context has pipeline_run_idNone by construction; pipeline_id_for_cutoff therefore yieldsUNKNOWN; exact workflow payload/context/run relationship proves the owning pipeline",
                           "storage_policy": "provider_registry.py:138-176 drops point; normalization_service.py persists binary_eligibleNone; catalyst feature computes binaryFalse; rejected guidance adds guidance_rows_rejected",
                           "opportunity_available_weights": ".25+.15+.10=.50 < .60; never zero-fill unavailable components"}

if __name__ == "__main__":
    dump(DEST / "ceri-expected.json", {"estimates": estimates, "guidance": guidance, "catalysts": catalysts,
         "revisions": revisions, "derived": derived, "price_response": prices, "captures": captures,
         "earnings": [], "changes": [], "alerts": []})
    dump(DEST / "ceri-derivations.json", {"per_ticker": derivations,
         "policy_sources": ["point_in_time_query.py:154-314,498-521", "revision_feature_service.py:194-396",
                            "guidance_comparison_service.py:17-25", "normalization_service.py:252-272",
                            "catalyst_taxonomy.py:42-72", "feature_rebuild_service.py:830-1020,1965-2034",
                            "opportunity_score_service.py:84-196", "confidence_service.py:52-158",
                            "contextual_effective_configuration.py:585-594", "change_detection_service.py:72-84,476-486"],
         "normalization": "Four immutable EPS observations per issuer; same-provider USD at unit scale; no corrections/supersession; refresh deduplicates source and normalized observations",
         "guidance": "The supplied SEC action RAISED is checked against prior guidance. No prior exists: native contract requires UNKNOWN/INSUFFICIENT, rejected GUIDANCE_ACTION_UNKNOWN. No accepted guidance derived feature or guidance change.",
         "catalysts": "Payload omits issuer_relevance, subject/title and confidence. Normalize subtype launch into subject launch, confidence Normal, relevance null. Unverified relevance rejects scoring and change generation; PRODUCT is not binary.",
         "freshness": "Operational ingestion completion remains live and later than fixed business cutoff. No completed provider checks are eligible at cutoff; feed freshness is unavailable, not stale or fabricated fresh. Source receipt timestamps remain independently known and eligible.",
         "absence": "No accepted reported earnings, guidance or issuer-verified catalyst; no causal price event. No prior certified target capture, so no financial score change. No eligible catalyst/guidance changes; alert population empty.",
         "ownership": "All pipeline feature/capture rows bind target ticker/company, post-acquisition CERI context at20:32, frozen config/version and semantic source keys. Generated IDs are resolved relationally.",
         "precision": "revision columns half-even six places, compared exactly; component/score fields absolute1e-6; nulls and semantic membership exact"})
    print("CERI independent: 40 estimates,30 revisions,20 derived,10 unrated captures; changes/alerts empty")
